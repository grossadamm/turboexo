cmake_minimum_required(VERSION 3.16)
project(turboexo_dashboard_host LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LVGL_SOURCE_DIR "" CACHE PATH "LVGL v9.5.0 source")
if(NOT LVGL_SOURCE_DIR AND DEFINED ENV{TURBOEXO_DEPS})
 set(LVGL_SOURCE_DIR "$ENV{TURBOEXO_DEPS}/lvgl-9.5.0")
endif()
if(NOT EXISTS "${LVGL_SOURCE_DIR}/lv_version.h")
 message(FATAL_ERROR "Set LVGL_SOURCE_DIR to the persistent LVGL v9.5.0 source tree")
endif()
file(READ "${LVGL_SOURCE_DIR}/lv_version.h" LV_VERSION_HEADER)
foreach(PART "MAJOR;9" "MINOR;5" "PATCH;0")
 list(GET PART 0 NAME)
 list(GET PART 1 VALUE)
 if(NOT LV_VERSION_HEADER MATCHES "#define LVGL_VERSION_${NAME} +${VALUE}([\r\n]|$)")
  message(FATAL_ERROR "LVGL source must be v9.5.0")
 endif()
endforeach()
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(LV_BUILD_CONF_PATH "${LV_CONF_PATH}" CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/lvgl")
set(DASHBOARD_FONTS ../assets/cluster_num_64.c ../assets/cluster_num_96.c ../assets/cluster_num_144.c ../assets/cluster_num_220.c ../assets/cluster_text_26.c ../assets/cluster_text_34.c)
add_executable(cluster_sim main.cpp ../ui/dashboard.cpp ../assets/cluster_icons.c ${DASHBOARD_FONTS} ../core/replay.cpp ../core/history.cpp)
target_include_directories(cluster_sim PRIVATE ../include)
target_link_libraries(cluster_sim PRIVATE lvgl)
find_package(SDL2 QUIET)
if(SDL2_FOUND)
 target_compile_definitions(cluster_sim PRIVATE CLUSTER_HAVE_SDL=1)
 target_link_libraries(cluster_sim PRIVATE SDL2::SDL2)
endif()
add_executable(rpm_animation_tests rpm_animation_tests.cpp)
target_include_directories(rpm_animation_tests PRIVATE ../include)
enable_testing()
add_test(NAME rpm_animation COMMAND rpm_animation_tests)
